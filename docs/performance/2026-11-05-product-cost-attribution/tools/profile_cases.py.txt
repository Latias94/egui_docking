import csv, hashlib, json, math, subprocess, sys
from datetime import datetime, timezone
from pathlib import Path
root = Path(__file__).resolve().parent
label = sys.argv[1]
metadata = json.loads((root / label / 'environment.json').read_text())
binary = root / (label + '-egui_dockspace')
if hashlib.sha256(binary.read_bytes()).hexdigest() != metadata['binary_sha256']['egui_dockspace']:
    raise RuntimeError('Frozen binary mismatch')
rows = list(csv.DictReader((root / label / 'summary.csv').open()))
for scenario in ['idle', 'surface_resize']:
    median = next(int(row['p50_ns']) for row in rows if row['product'] == 'egui_dockspace' and row['panes'] == '1024' and row['scenario'] == scenario)
    samples = max(128, math.ceil(12_000_000_000 / median))
    prefix = root / f'{label}-{scenario}'
    if prefix.with_name(prefix.name + '-profile-meta.json').exists():
        raise RuntimeError('Refusing to overwrite profile')
    command = [str(binary), '--samples', str(samples), '--warmup', '6', '--panes', '1024', '--scenario', scenario]
    meta = {'command': command, 'binary_sha256': metadata['binary_sha256']['egui_dockspace'], 'start_utc': datetime.now(timezone.utc).isoformat(), 'scope': 'First eight seconds of one isolated scenario, including readiness/warmup; separate from all formal timings. Sample counts are not wall-clock phase times.'}
    print('Profile', label, scenario, samples, flush=True)
    with prefix.with_name(prefix.name + '-profile.csv').open('w') as out, prefix.with_name(prefix.name + '-profile.stderr').open('w') as err:
        process = subprocess.Popen(command, stdout=out, stderr=err)
        meta['sample_command'] = ['sample', str(process.pid), '8', '-file', str(prefix.with_name(prefix.name + '-cpu.txt'))]
        with prefix.with_name(prefix.name + '-sampler.log').open('w') as log:
            meta['sampler_returncode'] = subprocess.run(meta['sample_command'], stdout=log, stderr=subprocess.STDOUT).returncode
        meta['returncode'] = process.wait()
    meta['end_utc'] = datetime.now(timezone.utc).isoformat()
    prefix.with_name(prefix.name + '-profile-meta.json').write_text(json.dumps(meta, indent=2) + '\n')
    if meta['returncode'] or meta['sampler_returncode']:
        raise RuntimeError('Profile process failed')
    with prefix.with_name(prefix.name + '-attribution.json').open('w') as output:
        subprocess.run([sys.executable, str(root / 'summarize_cpu.py'), str(prefix.with_name(prefix.name + '-cpu.txt'))], stdout=output, check=True)
    print('Complete', scenario, flush=True)
