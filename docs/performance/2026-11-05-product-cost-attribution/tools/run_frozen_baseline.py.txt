import csv, hashlib, importlib.util, json, subprocess
from pathlib import Path
from datetime import datetime, timezone
root = Path('/Users/frankorz/codes/rust/egui_docking')
out = Path('/tmp/dockspace-product-attribution-20261105-868e635/fresh-baseline')
out.mkdir(exist_ok=False)
spec = importlib.util.spec_from_file_location('comparison', root / 'scripts/compare_products.py')
comparison = importlib.util.module_from_spec(spec)
spec.loader.exec_module(comparison)
metadata = json.loads((root / 'docs/performance/2026-11-05-product-comparison/final/environment.json').read_text())
metadata['measured_binary_origin'] = 'Frozen executables from the previous final qualification; all original build input hashes are retained.'
metadata['started_utc'] = comparison.utc_now()
metadata['source_commit'] = '868e63586be5ecfd1647b62f3b06cc4c90eecdd3'
metadata['runs'] = []
metadata['binary_paths'] = {
    'egui_dockspace': '/tmp/dockspace-product-comparison-01a1a5fb/empty-final-egui_dockspace-product-cost',
    'dear_imgui': '/tmp/dockspace-product-comparison-01a1a5fb/empty-final-dear_imgui-product-cost',
}
for product, path in metadata['binary_paths'].items():
    if comparison.digest(Path(path)) != metadata['binary_sha256'][product]:
        raise RuntimeError('Frozen executable hash mismatch')
rows = []
for run, product in enumerate(['egui_dockspace', 'dear_imgui', 'dear_imgui', 'egui_dockspace'], 1):
    command = [metadata['binary_paths'][product], '--samples', '24', '--warmup', '6', '--panes', '16,128,1024']
    print('Run', run, product, flush=True)
    started = comparison.utc_now()
    result = subprocess.run(command, cwd=root, text=True, capture_output=True)
    (out / f'run-{run}-{product}.csv').write_text(result.stdout)
    (out / f'run-{run}-{product}.stderr').write_text(result.stderr)
    metadata['runs'].append({'run':run,'product':product,'command':command,'started_utc':started,'finished_utc':comparison.utc_now(),'returncode':result.returncode})
    (out / 'environment.json').write_text(json.dumps(metadata,indent=2)+'\n')
    if result.returncode: raise RuntimeError(result.stderr[-6000:])
    rows.extend({'run':run, **r} for r in comparison.validate_rows(result.stdout,product,[16,128,1024],24))
comparison.write_csv(out/'raw.csv',['run',*comparison.HEADER],rows)
summaries=[]
for product in comparison.PRODUCTS:
    for panes in [16,128,1024]:
        for scenario,frames in comparison.SCENARIOS.items():
            selected=[r for r in rows if r['product']==product and r['panes']==panes and r['scenario']==scenario]
            values=[r['elapsed_ns'] for r in selected]
            summary={'product':product,'panes':panes,'scenario':scenario,'frames_per_sample':frames,'samples':len(selected),'p50_ns':comparison.percentile(values,50),'p95_ns':comparison.percentile(values,95),'max_ns':max(values)}
            for key in ['passes','content_calls','visible_contents','vertices','indices','draw_commands']:
                summary['p50_'+key]=comparison.percentile([r[key] for r in selected],50)
            summaries.append(summary)
comparison.write_csv(out/'summary.csv',list(summaries[0]),summaries)
metadata['finished_utc']=comparison.utc_now()
metadata['results_sha256']={name:comparison.digest(out/name) for name in ['raw.csv','summary.csv']}
(out/'environment.json').write_text(json.dumps(metadata,indent=2)+'\n')
print('Complete fresh frozen baseline:',len(rows),'rows',flush=True)
