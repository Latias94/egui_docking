import csv, hashlib, importlib.util, json, subprocess
from pathlib import Path
root = Path(__file__).resolve().parent
repo = Path('/Users/frankorz/codes/rust/egui_docking')
out = root / 'confirmation'
out.mkdir(exist_ok=False)
spec = importlib.util.spec_from_file_location('comparison', repo / 'scripts/compare_products.py')
comparison = importlib.util.module_from_spec(spec)
spec.loader.exec_module(comparison)
records, commands = [], []
for run, condition in enumerate(['candidate-two', 'candidate-four', 'candidate-four', 'candidate-two'], 1):
    binary = root / (condition + '-egui_dockspace')
    environment = json.loads((root / condition / 'environment.json').read_text())
    digest = comparison.digest(binary)
    if digest != environment['binary_sha256']['egui_dockspace']:
        raise RuntimeError('Frozen binary mismatch')
    command = [str(binary), '--samples', '48', '--warmup', '6', '--panes', '1024']
    started = comparison.utc_now()
    print('Confirm', run, condition, flush=True)
    result = subprocess.run(command, cwd=repo, text=True, capture_output=True)
    (out / f'run-{run}-{condition}.csv').write_text(result.stdout)
    (out / f'run-{run}-{condition}.stderr').write_text(result.stderr)
    commands.append({'run':run,'condition':condition,'command':command,'binary_sha256':digest,'start_utc':started,'end_utc':comparison.utc_now(),'returncode':result.returncode})
    (out / 'environment.json').write_text(json.dumps({'scope':'Timing-only version confirmation. No build/test/sampler overlaps. Two/four/four/two; 48 samples, six warmups per case. Original build provenance remains in each condition environment.','runs':commands},indent=2)+'\n')
    if result.returncode:
        raise RuntimeError(result.stderr[-6000:])
    records.extend({'condition':condition,'run':run,**row} for row in comparison.validate_rows(result.stdout,'egui_dockspace',[1024],48))
comparison.write_csv(out/'raw.csv',['condition','run',*comparison.HEADER],records)
summaries=[]
for condition in ['candidate-two','candidate-four']:
    for scenario in comparison.SCENARIOS:
        values=[row['elapsed_ns'] for row in records if row['condition']==condition and row['scenario']==scenario]
        summaries.append({'condition':condition,'scenario':scenario,'samples':len(values),'p50_ns':comparison.percentile(values,50),'p95_ns':comparison.percentile(values,95),'max_ns':max(values)})
comparison.write_csv(out/'summary.csv',list(summaries[0]),summaries)
print('Complete interleaved version confirmation:',len(records),'rows',flush=True)
